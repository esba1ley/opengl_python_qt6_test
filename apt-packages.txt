# Debian / Raspberry Pi OS packages to run and develop this program.
# Same names on both; see README.md, "Raspberry Pi OS versus Debian".
#
#     sudo apt install $(sed 's/#.*//' apt-packages.txt)
#
# Verified against the Debian archive for bookworm (12) and trixie (13):
# python3-pyqt6 6.4.2 / 6.9.0, python3-opengl 3.1.6 / 3.1.9,
# python3-numpy 1.24 / 2.2. Environment.yml carries the same floors.

# Runtime
python3-pyqt6        # QtCore, QtGui, QtWidgets, QtOpenGL, QtOpenGLWidgets
python3-opengl       # PyOpenGL
python3-numpy
libgl1               # libGL.so.1, which PyOpenGL loads
libgl1-mesa-dri      # the Mesa drivers, V3D included; preinstalled on Pi OS desktop
qt6-wayland          # Qt's native Wayland plugin; Qt recommends it, so usually present

# Development
bumpversion          # bump2version, same tool under Debian's name
python3-venv         # for ruff and the docs tooling, which are not in apt or too old
python3-pip
